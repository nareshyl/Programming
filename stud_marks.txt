#include <stdio.h>

struct Student {
    int roll_no;
    char name[30];
    float marks;
};

/* Function to write student records */
void writeFile() {
    FILE *fp;
    struct Student s;
    int i;

    fp = fopen("transactions.dat", "w");

    if (fp == NULL) {
        printf("Error opening file!\n");
        return;
    }

    printf("Enter details of 3 students:\n");

    for (i = 0; i < 3; i++) {
        printf("\nStudent %d\n", i + 1);

        printf("Roll Number: ");
        scanf("%d", &s.roll_no);

        printf("Name: ");
        scanf(" %29[^\n]", s.name);

        printf("Marks: ");
        scanf("%f", &s.marks);

        fprintf(fp, "%d %s %.2f\n",
                s.roll_no, s.name, s.marks);
    }

    fclose(fp);

    printf("\nStudent records saved successfully.\n");
}

/* Function to read student records */
void readFile() {
    FILE *fp;
    struct Student s;

    fp = fopen("transactions.dat", "r");

    if (fp == NULL) {
        printf("Error opening file for reading!\n");
        return;
    }

    printf("\n-------------------------------------\n");
    printf("       STUDENT RECORDS\n");
    printf("-------------------------------------\n");
    printf("Roll No\tName\t\tMarks\n");
    printf("-------------------------------------\n");

    while (fscanf(fp, "%d %29s %f",
                  &s.roll_no, s.name, &s.marks) == 3) {

        printf("%d\t%-15s %.2f\n",
               s.roll_no, s.name, s.marks);
    }

    fclose(fp);
}

int main() {
    writeFile();
    readFile();

    return 0;
}
