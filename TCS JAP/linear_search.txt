#include <stdio.h>

int main() {
    int arr[] = {4, 2, 7, 1, 9};
    int n = 5;
    int target = 7;
    int i;
    int index = -1;

    // Linear Search
    for (i = 0; i < n; i++) {
        if (arr[i] == target) {
            index = i;
            break;
        }
    }

    if (index != -1) {
        printf("Element %d found at index %d\n", target, index);
    } else {
        printf("Element %d not found\n", target);
    }

    return 0;
}
