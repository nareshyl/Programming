#include <stdio.h>

// Size of each small run
#define RUN 4

// Insertion Sort for a small run
void insertionSort(int arr[], int left, int right) {
    int i, j, key;

    for (i = left + 1; i <= right; i++) {
        key = arr[i];
        j = i - 1;

        while (j >= left && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }

        arr[j + 1] = key;
    }
}

// Merge two sorted parts
void merge(int arr[], int left, int mid, int right) {
    int n1 = mid - left + 1;
    int n2 = right - mid;

    int L[n1], R[n2];

    int i, j, k;

    // Copy data into temporary arrays
    for (i = 0; i < n1; i++)
        L[i] = arr[left + i];

    for (j = 0; j < n2; j++)
        R[j] = arr[mid + 1 + j];

    i = 0;
    j = 0;
    k = left;

    // Merge the two arrays
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) {
            arr[k] = L[i];
            i++;
        } else {
            arr[k] = R[j];
            j++;
        }
        k++;
    }

    // Copy remaining elements of L[]
    while (i < n1) {
        arr[k] = L[i];
        i++;
        k++;
    }

    // Copy remaining elements of R[]
    while (j < n2) {
        arr[k] = R[j];
        j++;
        k++;
    }
}

// Simplified Tim Sort
void timSort(int arr[], int n) {
    int i, size, left, mid, right;

    // Step 1: Sort small runs using Insertion Sort
    for (i = 0; i < n; i += RUN) {
        int right = i + RUN - 1;

        if (right >= n)
            right = n - 1;

        insertionSort(arr, i, right);
    }

    // Step 2: Merge the sorted runs
    for (size = RUN; size < n; size = 2 * size) {

        for (left = 0; left < n; left += 2 * size) {

            mid = left + size - 1;
            right = left + 2 * size - 1;

            if (mid >= n - 1)
                continue;

            if (right >= n)
                right = n - 1;

            merge(arr, left, mid, right);
        }
    }
}

int main() {
    int arr[] = {3, 7, 4, 8, 6, 2, 1, 5};
    int n = sizeof(arr) / sizeof(arr[0]);

    printf("Original array: ");

    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }

    // Apply Tim Sort
    timSort(arr, n);

    printf("\nSorted array: ");

    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }

    return 0;
}
