#include <stdio.h>

#define MAX 100

// DFS function
int DFS(int graph[MAX][MAX],
        int vertices,
        int current,
        int visited[MAX],
        int recStack[MAX])
{
    // Mark current node as visited
    visited[current] = 1;

    // Add current node to recursion stack
    recStack[current] = 1;

    // Check all adjacent vertices
    for (int i = 0; i < vertices; i++)
    {
        if (graph[current][i] == 1)
        {
            // If node is not visited, continue DFS
            if (!visited[i])
            {
                if (DFS(graph,
                        vertices,
                        i,
                        visited,
                        recStack))
                {
                    return 1;
                }
            }

            // Node is already in current DFS path
            else if (recStack[i])
            {
                return 1;
            }
        }
    }

    // Remove node from recursion stack
    recStack[current] = 0;

    return 0;
}

// Detect cycle
int hasCycle(int graph[MAX][MAX], int vertices)
{
    int visited[MAX] = {0};
    int recStack[MAX] = {0};

    for (int i = 0; i < vertices; i++)
    {
        if (!visited[i])
        {
            if (DFS(graph,
                    vertices,
                    i,
                    visited,
                    recStack))
            {
                return 1;
            }
        }
    }

    return 0;
}

int main()
{
    int graph[MAX][MAX] = {0};

    int vertices = 4;

    /*
        Directed graph:

        0 → 1
        ↑   ↓
        3 ← 2

        Cycle:
        0 → 1 → 2 → 3 → 0
    */

    graph[0][1] = 1;
    graph[1][2] = 1;
    graph[2][3] = 1;
    graph[3][0] = 1;

    if (hasCycle(graph, vertices))
    {
        printf("Cycle detected in the graph.\n");
    }
    else
    {
        printf("No cycle detected in the graph.\n");
    }

    return 0;
}
