//Exercise 1 — Reverse a Stack Using Recursion
#include <stdio.h>

#define MAX 100

int stack[MAX];
int top = -1;

void push(int x)
{
    stack[++top] = x;
}

int pop()
{
    return stack[top--];
}

void insertBottom(int x)
{
    int temp;

    if (top == -1)
    {
        push(x);
        return;
    }

    temp = pop();
    insertBottom(x);
    push(temp);
}

void reverseStack()
{
    int temp;

    if (top <= 0)
        return;

    temp = pop();
    reverseStack();
    insertBottom(temp);
}

void display()
{
    int i;

    for (i = top; i >= 0; i--)
        printf("%d ", stack[i]);

    printf("\n");
}

int main()
{
    int n, i, x;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &x);
        push(x);
    }

    printf("Original stack: ");
    display();

    reverseStack();

    printf("Reversed stack: ");
    display();

    return 0;
}
