#include <stdio.h>
#include <limits.h>

#define N 4
#define INF 999999

int dist[N][N] = {
    {0, 10, 15, 20},
    {10, 0, 35, 25},
    {15, 35, 0, 30},
    {20, 25, 30, 0}
};

int dp[1 << N][N];

// Minimum of two numbers
int min(int a, int b) {
    return (a < b) ? a : b;
}

// TSP using DP + Bitmasking
int tsp(int mask, int pos) {

    // All cities visited
    if (mask == (1 << N) - 1) {
        return dist[pos][0];
    }

    // Already calculated
    if (dp[mask][pos] != -1) {
        return dp[mask][pos];
    }

    int answer = INF;

    // Try visiting every unvisited city
    for (int city = 0; city < N; city++) {

        if ((mask & (1 << city)) == 0) {

            int newCost = dist[pos][city]
                        + tsp(mask | (1 << city), city);

            answer = min(answer, newCost);
        }
    }

    dp[mask][pos] = answer;

    return answer;
}

int main() {

    // Initialize DP table
    for (int i = 0; i < (1 << N); i++) {
        for (int j = 0; j < N; j++) {
            dp[i][j] = -1;
        }
    }

    // Start from City 0
    int minimumCost = tsp(1, 0);

    printf("Minimum TSP Cost = %d\n", minimumCost);

    return 0;
}
