#include <stdio.h>

#define MAX 100

struct Graph
{
    int vertices;
    int adj[MAX][MAX];
};

// Initialize graph
void initGraph(struct Graph *g)
{
    g->vertices = 0;

    for (int i = 0; i < MAX; i++)
    {
        for (int j = 0; j < MAX; j++)
        {
            g->adj[i][j] = 0;
        }
    }
}

// Add Vertex
void addVertex(struct Graph *g)
{
    if (g->vertices >= MAX)
    {
        printf("Maximum number of vertices reached.\n");
        return;
    }

    g->vertices++;

    printf("Vertex %d added.\n", g->vertices - 1);
}

// Add Edge
void addEdge(struct Graph *g, int u, int v)
{
    if (u >= g->vertices || v >= g->vertices)
    {
        printf("Invalid vertex.\n");
        return;
    }

    // Undirected graph
    g->adj[u][v] = 1;
    g->adj[v][u] = 1;

    printf("Edge added between %d and %d.\n", u, v);
}

// Display Graph
void display(struct Graph *g)
{
    printf("\n===== GRAPH =====\n");

    for (int i = 0; i < g->vertices; i++)
    {
        printf("%d -> ", i);

        for (int j = 0; j < g->vertices; j++)
        {
            if (g->adj[i][j] == 1)
            {
                printf("%d ", j);
            }
        }

        printf("\n");
    }
}

// BFS Traversal
void BFS(struct Graph *g, int start)
{
    int visited[MAX] = {0};
    int queue[MAX];

    int front = 0;
    int rear = 0;

    queue[rear++] = start;
    visited[start] = 1;

    printf("\nBFS Traversal: ");

    while (front < rear)
    {
        int current = queue[front++];

        printf("%d ", current);

        for (int i = 0; i < g->vertices; i++)
        {
            if (g->adj[current][i] == 1 &&
                visited[i] == 0)
            {
                visited[i] = 1;
                queue[rear++] = i;
            }
        }
    }

    printf("\n");
}

// DFS Traversal
void DFSUtil(struct Graph *g, int vertex, int visited[])
{
    visited[vertex] = 1;

    printf("%d ", vertex);

    for (int i = 0; i < g->vertices; i++)
    {
        if (g->adj[vertex][i] == 1 &&
            visited[i] == 0)
        {
            DFSUtil(g, i, visited);
        }
    }
}

void DFS(struct Graph *g, int start)
{
    int visited[MAX] = {0};

    printf("\nDFS Traversal: ");

    DFSUtil(g, start, visited);

    printf("\n");
}


// Main Function
int main()
{
    struct Graph g;

    initGraph(&g);

    // Add vertices
    for (int i = 0; i < 6; i++)
    {
        addVertex(&g);
    }

    // Add edges
    addEdge(&g, 0, 1);
    addEdge(&g, 0, 2);
    addEdge(&g, 1, 3);
    addEdge(&g, 1, 4);
    addEdge(&g, 2, 5);

    // Display graph
    display(&g);

    // BFS starting from vertex 0
    BFS(&g, 0);

    // DFS starting from vertex 0
    DFS(&g, 0);

    return 0;
}
