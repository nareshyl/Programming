#include <stdio.h>

int binarySearch(int arr[], int n, int target) {
    int low = 0;
    int high = n - 1;

    while (low <= high) {
        int mid = low + (high - low) / 2;

        if (arr[mid] == target) {
            return mid;
        }
        else if (arr[mid] < target) {
            low = mid + 1;
        }
        else {
            high = mid - 1;
        }
    }

    return -1;
}

int main() {
    int arr[] = {1, 3, 5, 7, 8, 9};
    int n = 6;
    int target = 8;

    int result = binarySearch(arr, n, target);

    if (result != -1) {
        printf("%d found at index %d\n", target, result);
    } else {
        printf("%d not found\n", target);
    }

    return 0;
}
