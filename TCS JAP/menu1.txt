#include <stdio.h>
#include <stdlib.h>

/* ---------- Pointer Basics ---------- */

// Swap two numbers using pointers
void swap(int *a, int *b)
{
    int temp = *a;
    *a = *b;
    *b = temp;
}

// Modify a value using a pointer
void modifyValue(int *p)
{
    *p = *p + 10;
}


/* ---------- Pointer Arithmetic ---------- */

// Display array using pointer arithmetic
void pointerArithmetic(int *arr, int size)
{
    printf("\nArray using pointer arithmetic:\n");

    for (int i = 0; i < size; i++)
    {
        printf("Address: %p  Value: %d\n",
               (void *)(arr + i), *(arr + i));
    }
}


/* ---------- Pointers and Arrays ---------- */

// Reverse array using pointers
void reverseArray(int *arr, int size)
{
    int *start = arr;
    int *end = arr + size - 1;

    while (start < end)
    {
        int temp = *start;
        *start = *end;
        *end = temp;

        start++;
        end--;
    }
}

// Display array using pointers
void displayArray(int *arr, int size)
{
    printf("Array: ");

    for (int i = 0; i < size; i++)
    {
        printf("%d ", *(arr + i));
    }

    printf("\n");
}


/* ---------- Pointers to Functions ---------- */

int add(int a, int b)
{
    return a + b;
}

int subtract(int a, int b)
{
    return a - b;
}

int multiply(int a, int b)
{
    return a * b;
}

int divideNumbers(int a, int b)
{
    if (b == 0)
    {
        printf("Error: Division by zero!\n");
        return 0;
    }

    return a / b;
}


/* ---------- Main Function ---------- */

int main()
{
    int choice;

    do
    {
        printf("\n========================================\n");
        printf("       POINTER PROBLEM SOLVING SYSTEM\n");
        printf("========================================\n");

        printf("1. Pointer Basics\n");
        printf("2. Pointer Arithmetic\n");
        printf("3. Pointers and Arrays\n");
        printf("4. Pointers to Functions\n");
        printf("5. Exit\n");

        printf("========================================\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);


        /* ---------- Pointer Basics ---------- */

        if (choice == 1)
        {
            int a, b;

            printf("\n--- Pointer Basics ---\n");

            printf("Enter first number: ");
            scanf("%d", &a);

            printf("Enter second number: ");
            scanf("%d", &b);

            printf("\nBefore swapping:\n");
            printf("a = %d\n", a);
            printf("b = %d\n", b);

            swap(&a, &b);

            printf("\nAfter swapping:\n");
            printf("a = %d\n", a);
            printf("b = %d\n", b);

            printf("\nAddresses:\n");
            printf("&a = %p\n", (void *)&a);
            printf("&b = %p\n", (void *)&b);
        }


        /* ---------- Pointer Arithmetic ---------- */

        else if (choice == 2)
        {
            int arr[5];

            printf("\n--- Pointer Arithmetic ---\n");

            printf("Enter 5 elements:\n");

            for (int i = 0; i < 5; i++)
            {
                scanf("%d", &arr[i]);
            }

            pointerArithmetic(arr, 5);
        }


        /* ---------- Pointers and Arrays ---------- */

        else if (choice == 3)
        {
            int size;

            printf("\n--- Pointers and Arrays ---\n");

            printf("Enter array size: ");
            scanf("%d", &size);

            if (size <= 0)
            {
                printf("Invalid array size!\n");
                continue;
            }

            int *arr = (int *)malloc(size * sizeof(int));

            if (arr == NULL)
            {
                printf("Memory allocation failed!\n");
                continue;
            }

            printf("Enter %d elements:\n", size);

            for (int i = 0; i < size; i++)
            {
                *(arr + i) = i + 1;
            }

            printf("\nOriginal ");
            displayArray(arr, size);

            reverseArray(arr, size);

            printf("Reversed ");
            displayArray(arr, size);

            free(arr);
            arr = NULL;
        }


        /* ---------- Pointers to Functions ---------- */

        else if (choice == 4)
        {
            int a, b;
            int operation;
            int (*operationFunction)(int, int);

            printf("\n--- Pointers to Functions ---\n");

            printf("Enter first number: ");
            scanf("%d", &a);

            printf("Enter second number: ");
            scanf("%d", &b);

            printf("\n1. Addition\n");
            printf("2. Subtraction\n");
            printf("3. Multiplication\n");
            printf("4. Division\n");

            printf("Choose operation: ");
            scanf("%d", &operation);

            switch (operation)
            {
                case 1:
                    operationFunction = add;
                    break;

                case 2:
                    operationFunction = subtract;
                    break;

                case 3:
                    operationFunction = multiply;
                    break;

                case 4:
                    operationFunction = divideNumbers;
                    break;

                default:
                    printf("Invalid operation!\n");
                    continue;
            }

            printf("\nResult = %d\n",
                   operationFunction(a, b));
        }


        /* ---------- Exit ---------- */

        else if (choice == 5)
        {
            printf("\nProgram terminated successfully.\n");
        }

        else

            printf("\nInvalid choice! Please try again.\n");
        }

    } while (choice != 5);

}
