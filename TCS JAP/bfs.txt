#include <stdio.h>

#define MAX 100

// Add an edge to the graph
void addEdge(int graph[MAX][MAX], int u, int v)
{
    graph[u][v] = 1;
    graph[v][u] = 1;
}

// BFS to find shortest path
void shortestPath(int graph[MAX][MAX],
                  int n,
                  int start,
                  int end)
{
    int queue[MAX];
    int front = 0;
    int rear = 0;

    int visited[MAX] = {0};
    int parent[MAX];

    int i;

    // Initialize parent array
    for (i = 0; i < n; i++)
    {
        parent[i] = -1;
    }

    // Start BFS
    queue[rear++] = start;
    visited[start] = 1;

    while (front < rear)
    {
        int current = queue[front++];

        // Destination found
        if (current == end)
        {
            break;
        }

        // Visit adjacent nodes
        for (i = 0; i < n; i++)
        {
            if (graph[current][i] == 1 && !visited[i])
            {
                visited[i] = 1;

                // Store previous node
                parent[i] = current;

                queue[rear++] = i;
            }
        }
    }

    // No path exists
    if (!visited[end])
    {
        printf("[]\n");
        return;
    }

    // Reconstruct path
    int path[MAX];
    int pathSize = 0;

    int current = end;

    while (current != -1)
    {
        path[pathSize++] = current;
        current = parent[current];
    }

    // Print path in correct order
    printf("Shortest Path: ");

    for (i = pathSize - 1; i >= 0; i--)
    {
        printf("%d", path[i]);

        if (i != 0)
        {
            printf(" -> ");
        }
    }

    printf("\n");
}

int main()
{
    int graph[MAX][MAX] = {0};

    int n = 6;

    /*
        Graph:

        0 ----- 1
        |       |
        |       |
        2 ----- 3 ----- 4
                        |
                        5
    */

    addEdge(graph, 0, 1);
    addEdge(graph, 0, 2);
    addEdge(graph, 1, 3);
    addEdge(graph, 2, 3);
    addEdge(graph, 3, 4);
    addEdge(graph, 4, 5);

    int start = 0;
    int end = 5;

    shortestPath(graph, n, start, end);

    return 0;
}
