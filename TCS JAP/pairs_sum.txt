//1.Find all pairs with the given sum
#include <stdio.h>

int main() {
    int arr[] = {1, 5, 7, -1, 5};
    int n = 5;
    int sum = 6;

    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {

            if (arr[i] + arr[j] == sum) {
                printf("(%d, %d) ", arr[i], arr[j]);
            }
        }
    }

    return 0;
}
