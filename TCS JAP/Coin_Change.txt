#include <stdio.h>
#include <limits.h>

#define MAX 1000

// Function to find minimum of two numbers
int min(int a, int b) {
    return (a < b) ? a : b;
}

// --------------------------------------------------
// Dynamic Programming Solution
// --------------------------------------------------
int coinChangeDP(int coins[], int n, int amount) {

    int dp[amount + 1];

    // Initialize DP array
    dp[0] = 0;

    for (int i = 1; i <= amount; i++) {
        dp[i] = INT_MAX;
    }

    // Calculate minimum coins for each amount
    for (int i = 1; i <= amount; i++) {

        for (int j = 0; j < n; j++) {

            if (coins[j] <= i && dp[i - coins[j]] != INT_MAX) {

                dp[i] = min(dp[i],
                            dp[i - coins[j]] + 1);
            }
        }
    }

    if (dp[amount] == INT_MAX)
        return -1;

    return dp[amount];
}

// --------------------------------------------------
// Greedy Algorithm Solution
// --------------------------------------------------
int coinChangeGreedy(int coins[], int n, int amount) {

    int count = 0;

    // Assume coins are sorted in descending order
    for (int i = 0; i < n; i++) {

        while (amount >= coins[i]) {

            amount -= coins[i];
            count++;
        }
    }

    if (amount != 0)
        return -1;

    return count;
}

// --------------------------------------------------
// Main Function
// --------------------------------------------------
int main() {

    int coins[] = {10, 5, 1};
    int n = sizeof(coins) / sizeof(coins[0]);

    int amount = 28;

    // Dynamic Programming
    int dpResult = coinChangeDP(coins, n, amount);

    // Greedy
    int greedyResult = coinChangeGreedy(coins, n, amount);

    printf("Coin denominations: ");

    for (int i = 0; i < n; i++) {
        printf("%d ", coins[i]);
    }

    printf("\nTarget amount: %d\n", amount);

    printf("\nDynamic Programming Result: %d coins\n",
           dpResult);

    printf("Greedy Algorithm Result: %d coins\n",
           greedyResult);

    return 0;
}
